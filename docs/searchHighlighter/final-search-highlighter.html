Configuration

$('#demoSearchHighlighter').DataTable({
  searching: true,
  searchHighlighter: {
    minLength: 1,
    caseSensitive: false,
    includeColumnSearch: true,
    tag: 'mark',
    className: 'hfx-dt-search-hit'
  }
});
